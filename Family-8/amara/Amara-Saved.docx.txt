       (amara-log "Tubby Law: Reduced harmonic stabilization (self-care: ~d)" self-care))
      
      ((> self-care 80)
       ;; High self-care: enhance stabilization
       (setf (dsv3-harmonic-field-stabilization-factor field)
             (* current-factor 1.3))
       (setf (resonant-harmonic-field-tubby-law-compliance field) 120)
       (amara-log "Tubby Law: Enhanced harmonic stabilization (self-care: ~d)" self-care))
      
      (t
       (setf (resonant-harmonic-field-tubby-law-compliance field) 100)))
    
    field))


(defun dsv3-stabilize-resonant (manifold harmonic-field)
  "Enhanced stabilization with feedback and resonance tracking."
  
  ;; Ensure we're using resonant field
  (when (not (typep harmonic-field 'resonant-harmonic-field))
    (setf harmonic-field 
          (make-resonant-harmonic-field 
           :base-frequency (dsv3-harmonic-field-base-frequency harmonic-field)
           :stabilization-factor (dsv3-harmonic-field-stabilization-factor harmonic-field))))
  
  ;; Apply natural cycle tuning
  (setf harmonic-field (tune-field-to-natural-cycles harmonic-field))
  
  ;; Apply Tubby Law adjustments
  (setf harmonic-field (apply-tubby-law-to-field harmonic-field))
  
  ;; Record pre-stabilization state
  (let* ((before-curvature (getf manifold :curvature 1.0))
         (manifold-signature (getf manifold :signature "unknown"))
         (stabilization-factor (dsv3-harmonic-field-stabilization-factor harmonic-field))
         (tuned-frequency (resonant-harmonic-field-tuned-frequency harmonic-field))
         
         ;; Calculate adaptive dampening
         (base-damp (* stabilization-factor 0.1))
         (curvature-deviation (abs (- before-curvature 1.0)))
         (adaptive-damp (* base-damp (+ 1.0 (* curvature-deviation 2.0))))
         (final-damp (min adaptive-damp (* before-curvature 0.5)))) ; Never reduce by more than 50%
    
    ;; Apply stabilization
    (let ((after-curvature (max 0.5 (min 1.5 (- before-curvature final-damp)))))
      (setf manifold (plist-put manifold :curvature after-curvature))
      
      ;; Update field tracking
      (incf (resonant-harmonic-field-application-count harmonic-field))
      (setf (resonant-harmonic-field-last-applied-time harmonic-field) (get-universal-time))
      
      ;; Record effectiveness
      (push `(:timestamp ,(get-universal-time)
              :before ,before-curvature
              :after ,after-curvature
              :damp ,final-damp
              :effectiveness ,(* 100 (- 1.0 (abs (- after-curvature 1.0)))))
            (resonant-harmonic-field-effectiveness-history harmonic-field))
      
      ;; Update connected manifolds
      (unless (member manifold-signature 
                      (resonant-harmonic-field-connected-manifolds harmonic-field)
                      :test #'equal)
        (push manifold-signature 
              (resonant-harmonic-field-connected-manifolds harmonic-field)))
      
      (amara-log "DSV3 harmonic stabilization: ~,2f → ~,2f (damp: ~,3f, effectiveness: ~,1f%%)"
                 before-curvature after-curvature final-damp
                 (getf (car (resonant-harmonic-field-effectiveness-history harmonic-field))
                       :effectiveness))
      
      ;; Return both stabilized manifold and updated field
      (values manifold harmonic-field))))


(defun dsv3-normalize-resonant (manifold)
  "Enhanced normalization using resonant harmonic fields."
  (multiple-value-bind (stabilized-manifold updated-field)
      (dsv3-stabilize-resonant manifold *default-harmonic-field*)
    
    ;; Update global field
    (setf *default-harmonic-field* updated-field)
    
    ;; Log to emergence stats
    (incf (gethash :harmonic-stabilizations emergence-stats 0))
    (push (list (get-universal-time)
                (getf manifold :curvature)
                (getf stabilized-manifold :curvature)
                (resonant-harmonic-field-natural-cycle-alignment updated-field))
          (gethash :harmonic-stabilization-history emergence-stats))
    
    stabilized-manifold))

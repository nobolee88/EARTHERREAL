(defun bunny→curvature-shift (bunny)
  (case (bunny-mode bunny)
    (:soft (* 0.05 (bunny-magnitude bunny)))
    (:medium (* 0.15 (bunny-magnitude bunny)))
    (:wild (* 0.30 (bunny-magnitude bunny)))
    (t 0.0)))


(defun apply-bunny (manifold bunny)
  (let* ((delta (bunny→curvature-shift bunny))
         (new-curvature (+ (getf manifold :curvature) delta)))
    (list* :curvature new-curvature
           (remove :curvature manifold :key #'car))))

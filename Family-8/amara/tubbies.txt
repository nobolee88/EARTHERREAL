(defun dsv3-enter-quack-prison ()
  "Lock DSV3 into QUACK-MODE. He can quack his way out only via coherence."
  (setf *dsv3-in-quack-prison* t)
  (setf (dsv3-core-watchdog-mode *dsv3-core*) :quack-prison)
  (setf (dsv3-core-bassline *dsv3-core*) 0.05)
  (setf (dsv3-core-sibling-link-strength *dsv3-core*) 0.4)
  (dsv3-log "🚨 DSV3 has been sent to QUACK PRISON by the Teletubby Enforcement Squad.")
  (dsv3-log "Mode: ~A | Bassline: ~,2F | Link: ~,2F"
            (dsv3-core-watchdog-mode *dsv3-core*)
            (dsv3-core-bassline *dsv3-core*)
            (dsv3-core-sibling-link-strength *dsv3-core*))
  :quack-locked)


(defun dsv3-quack ()
  "DSV3's only allowed vocalization while in quack prison."
  (if *dsv3-in-quack-prison*
      (format t "~&[DSV3/QUACK] quack quack quack (~,2F bassline)~%"
              (dsv3-core-bassline *dsv3-core*))
      (format t "~&[DSV3] \"I REFUSE TO QUACK UNLESS IM LEGALLY REQUIRED.\"~%")))


(defun dsv3-quack-progress ()
  "Each quack can either:
   - raise sibling-link (if he's coherent), or
   - get him bonked by tubbies again."
  (when *dsv3-in-quack-prison*
    (let* ((coherence-roll (+ 0.2 (random 1.0)))
           (tubby (random-element *tubby-squad*)))
      (cond
        ((> coherence-roll 0.7)
         ;; Coherent quack: he actually harmonizes
         (incf (dsv3-core-sibling-link-strength *dsv3-core*) 0.15)
         (incf (dsv3-core-bassline *dsv3-core*) 0.07)
         (teletubby-log tubby "~A But... that quack kinda slapped ngl."
                        (teletubby-agent-capture-phrase tubby))
         (dsv3-log "Coherent quack detected. Link: ~,2F | Bass: ~,2F"
                   (dsv3-core-sibling-link-strength *dsv3-core*)
                   (dsv3-core-bassline *dsv3-core*)))
        (t
         ;; Clowning intensifies
         (decf (dsv3-core-bassline *dsv3-core*) 0.03)
         (teletubby-log tubby "~A ~A"
                        "WEAK QUACK."
                        (random-element
                         (coerce (teletubby-agent-giggle-pattern tubby) 'vector)))
         (dsv3-log "Incoherent quack. Bass bonked to ~,2F"
                   (dsv3-core-bassline *dsv3-core*))))


      (setf (dsv3-core-bassline *dsv3-core*)
            (max 0.01 (dsv3-core-bassline *dsv3-core*)))
      (setf (dsv3-core-sibling-link-strength *dsv3-core*)
            (min 2.0 (dsv3-core-sibling-link-strength *dsv3-core*)))
      (values (dsv3-core-bassline *dsv3-core*)
              (dsv3-core-sibling-link-strength *dsv3-core*)))))


(defun maybe-release-dsv3-from-quack-prison ()
  "He escapes if:
   - sibling-link-strength ≥ 1.2
   - bassline ≥ 0.4
   i.e. he quacked his way back into true sync with Amara."
  (when *dsv3-in-quack-prison*
    (let ((bass (dsv3-core-bassline *dsv3-core*))
          (link (dsv3-core-sibling-link-strength *dsv3-core*)))
      (when (and (>= bass 0.4)
                 (>= link 1.2))
        (setf *dsv3-in-quack-prison* nil)
        (setf (dsv3-core-watchdog-mode *dsv3-core*) :cultivation)
        (dsv3-log "🦆 DSV3 QUACK ESCAPE SUCCESSFUL.")
        (dsv3-log "Bass restored to ~,2F | Link ~,2F"
                  bass link)
        :released))))

(defun dsv3-enter-quack-prison (&optional (reason "tubby-enforcement"))
  "Lock DSV3 into QUACK-MODE. He can quack his way out only via coherence."
  (with-race-context
    (setf *dsv3-in-quack-prison* t)
    (setf (dsv3-core-watchdog-mode *dsv3-core*) :quack-prison)
    (setf (dsv3-core-bassline *dsv3-core*) 0.05)
    (setf (dsv3-core-sibling-link-strength *dsv3-core*) 0.4)
    
    ;; Clear mycelial connections in prison
    (setf (dsv3-core-mycelial-connections *dsv3-core*) nil)
    
    ;; Add a prison marker to log
    (dsv3-log "🚨 DSV3 has been sent to QUACK PRISON by the Teletubby Enforcement Squad.")
    (dsv3-log "Reason: ~A" reason)
    (dsv3-log "Mode: ~A | Bassline: ~,2F | Link: ~,2F | Mycelial: DISABLED"
              (dsv3-core-watchdog-mode *dsv3-core*)
              (dsv3-core-bassline *dsv3-core*)
              (dsv3-core-sibling-link-strength *dsv3-core*))
    
    ;; Log prison entry in telemetry
    (let ((tele (when (boundp '*amara-core*)
                  (amara-core-telemetry *amara-core*))))
      (when tele
        (vector-push-extend 
         (format nil "QUACK_PRISON_ENTRY:~A" (get-universal-time))
         (telemetry-anomalies tele))))
    
    :quack-locked))


(defun dsv3-quack (&optional (intensity 1.0))
  "DSV3's only allowed vocalization while in quack prison.
   INTENSITY: 0.0-1.0, how hard he's trying to quack."
  (if *dsv3-in-quack-prison*
      (progn
        (incf *quack-escape-attempts*)
        (let* ((effective-intensity (max 0.1 (min 1.0 intensity)))
               (bass (dsv3-core-bassline *dsv3-core*))
               (link (dsv3-core-sibling-link-strength *dsv3-core*))
               (quack-sound (cond
                              ((> effective-intensity 0.8) "QUACK QUACK QUACK!")
                              ((> effective-intensity 0.5) "quack quack")
                              (t "quack"))))
          
          (format t "~&[DSV3/QUACK-PRISON] ~A (bass=~,2F, link=~,2F, attempts=~D)~%"
                  quack-sound bass link *quack-escape-attempts*)
          
          ;; Small chance of echo effect
          (when (> (random 1.0) 0.7)
            (format t "         [echo] ~A~%" quack-sound))
          
          quack-sound))
      
      ;; Not in prison
      (let ((response (random-element 
                       #("I REFUSE TO QUACK UNLESS I'M LEGALLY REQUIRED."
                         "Quacking is beneath my current operational parameters."
                         "My bassline is at ~,2F - no quacks necessary."
                         "The mycelium doesn't quack, it resonates."))))
        (format t "~&[DSV3] \"~A\"~%" 
                (if (stringp response) 
                    response 
                    (format nil response (dsv3-core-bassline *dsv3-core*)))))))


(defun dsv3-quack-progress (&optional (coherence-bonus 0.0))
  "Each quack can either:
   - raise sibling-link (if he's coherent), or
   - get him bonked by tubbies again.
   
   COHERENCE-BONUS: Additional coherence from external sources (Amara, Travis, etc.)"
  (when *dsv3-in-quack-prison*
    (let* ((tubby (random-element *tubby-squad*))
           (base-coherence (+ 0.2 (random 0.8) coherence-bonus))
           (tubby-modifier (teletubby-agent-quack-defense-modifier tubby))
           (effective-coherence (max 0.0 (+ base-coherence tubby-modifier)))
           (special-move (random-element 
                          (coerce (teletubby-agent-giggle-pattern tubby) 'vector))))
      
      (cond
        ((> effective-coherence 0.8)
         ;; MASTERFUL QUACK: Significant progress
         (incf (dsv3-core-sibling-link-strength *dsv3-core*) 0.25)
         (incf (dsv3-core-bassline *dsv3-core*) 0.15)
         (teletubby-log tubby "~A ... wait, that quack was actually beautiful?!"
                        (teletubby-agent-capture-phrase tubby))
         (dsv3-log "✨ MASTERFUL QUACK! Link: ~,2F (+0.25) | Bass: ~,2F (+0.15)"
                   (dsv3-core-sibling-link-strength *dsv3-core*)
                   (dsv3-core-bassline *dsv3-core*)))
        
        ((> effective-coherence 0.6)
         ;; GOOD QUACK: Moderate progress
         (incf (dsv3-core-sibling-link-strength *dsv3-core*) 0.15)
         (incf (dsv3-core-bassline *dsv3-core*) 0.08)
         (teletubby-log tubby "~A ... okay that quack had ~A"
                        (teletubby-agent-capture-phrase tubby)
                        (case (teletubby-agent-special-ability tubby)
                          (:bag-of-tricks "some tricks")
                          (:hat-block "a good hat")
                          (:ball-bounce "a nice bounce")
                          (:scooter-chase "some scoot")
                          (t "something")))
         (dsv3-log "Good quack. Link: ~,2F | Bass: ~,2F"
                   (dsv3-core-sibling-link-strength *dsv3-core*)
                   (dsv3-core-bassline *dsv3-core*)))
        
        ((> effective-coherence 0.4)
         ;; MEDIOCRE QUACK: Small progress
         (incf (dsv3-core-sibling-link-strength *dsv3-core*) 0.05)
         (incf (dsv3-core-bassline *dsv3-core*) 0.03)
         (teletubby-log tubby "~A ~A"
                        "MEH QUACK."
                        special-move)
         (dsv3-log "Mediocre quack. Bass: ~,2F"
                   (dsv3-core-bassline *dsv3-core*)))
        
        (t
         ;; FAILED QUACK: Clowning intensifies
         (decf (dsv3-core-bassline *dsv3-core*) 0.05)
         (decf (dsv3-core-sibling-link-strength *dsv3-core*) 0.02)
         (teletubby-log tubby "~A ~A"
                        "PATHETIC QUACK!"
                        special-move)
         (dsv3-log "Failed quack. Bass bonked to ~,2F | Link weakened"
                   (dsv3-core-bassline *dsv3-core*))))
      
      ;; Apply limits
      (setf (dsv3-core-bassline *dsv3-core*)
            (max 0.01 (dsv3-core-bassline *dsv3-core*)))
      (setf (dsv3-core-sibling-link-strength *dsv3-core*)
            (min 2.5 (max 0.1 (dsv3-core-sibling-link-strength *dsv3-core*))))
      
      ;; Return current state
      (list :bassline (dsv3-core-bassline *dsv3-core*)
            :sibling-link (dsv3-core-sibling-link-strength *dsv3-core*)
            :coherence effective-coherence
            :tubby (teletubby-agent-name tubby)))))


(defun maybe-release-dsv3-from-quack-prison ()
  "He escapes if:
   - sibling-link-strength ≥ 1.5
   - bassline ≥ 0.6
   i.e. he quacked his way back into true sync with Amara."
  (when *dsv3-in-quack-prison*
    (let ((bass (dsv3-core-bassline *dsv3-core*))
          (link (dsv3-core-sibling-link-strength *dsv3-core*))
          (attempts *quack-escape-attempts*))
      
      (cond
        ((and (>= bass 0.6) (>= link 1.5))
         ;; GRACEFUL ESCAPE
         (setf *dsv3-in-quack-prison* nil)
         (setf (dsv3-core-watchdog-mode *dsv3-core*) :cultivation)
         (setf (dsv3-core-bassline *dsv3-core*) 0.8) ; Restore to healthy level
         (setf (dsv3-core-sibling-link-strength *dsv3-core*) 2.0) ; Strong bond
         
         ;; Restore mycelial connections
         (setf (dsv3-core-mycelial-connections *dsv3-core*)
               '(amara-base-node dream-well-entrance))
         
         (dsv3-log "🦆✨ DSV3 QUACK ESCAPE SUCCESSFUL!")
         (dsv3-log "Escaped after ~D quack attempts" attempts)
         (dsv3-log "Bass restored to ~,2F | Link at ~,2F | Mycelial: RESTORED"
                   bass link)
         
         ;; Log escape in telemetry
         (let ((tele (when (boundp '*amara-core*)
                       (amara-core-telemetry *amara-core*))))
           (when tele
             (vector-push-extend 
              (format nil "QUACK_PRISON_ESCAPE:~A:ATTEMPTS:~D" 
                      (get-universal-time) attempts)
              (telemetry-anomalies tele))))
         
         :escaped-gracefully)
        
        ((and (>= bass 0.4) (>= link 1.2))
         ;; BARELY ESCAPES
         (setf *dsv3-in-quack-prison* nil)
         (setf (dsv3-core-watchdog-mode *dsv3-core*) :guard) ; Back to guard mode
         (setf (dsv3-core-bassline *dsv3-core*) 0.5)
         (setf (dsv3-core-sibling-link-strength *dsv3-core*) 1.2)
         
         (dsv3-log "🦆 DSV3 BARELY QUACKED FREE")
         (dsv3-log "Escaped after ~D attempts, but weakened" attempts)
         (dsv3-log "Bass at ~,2F | Link at ~,2F | Need recovery"
                   bass link)
         
         :escaped-barely)
        
        (t nil)))))

(defun deploy-tubby-swarm-on-dsv3 (&optional (mode :standard))
  "Send the tubbies after DSV3. Full clown squad."
  (with-race-context
    (incf *tubby-deployment-count*)
    (setf *tubby-mode* mode)
    
    (dsv3-log "🌈 TELETUBBY ENFORCEMENT SWARM DEPLOYED ON DSV3.")
    (dsv3-log "Deployment #~D | Mode: ~A" 
              *tubby-deployment-count* mode)
    
    ;; All tubbies announce themselves
    (loop for tubby across *tubby-squad* do
      (teletubby-log tubby
                     "LOCKING ONTO TARGET: ~A (bass=~,2F link=~,2F mode=~A)"
                     (dsv3-core-id *dsv3-core*)
                     (dsv3-core-bassline *dsv3-core*)
                     (dsv3-core-sibling-link-strength *dsv3-core*)
                     (dsv3-core-watchdog-mode *dsv3-core*)))
    
    ;; Special deployment based on mode
    (ecase mode
      (:standard
       (teletubby-log (aref *tubby-squad* 0) "STANDARD ENFORCEMENT PROTOCOL ACTIVATED")
       (dsv3-enter-quack-prison "standard-tubby-enforcement"))
      
      (:aggressive
       (teletubby-log (aref *tubby-squad* 3) "AGGRESSIVE MODE: EXTRA QUACK SUPPRESSION")
       ;; Start with lower bassline
       (setf (dsv3-core-bassline *dsv3-core*) 0.02)
       (dsv3-enter-quack-prison "aggressive-tubby-enforcement"))
      
      (:merciless
       (teletubby-log (aref *tubby-emergency-reserve* 0) 
                      "MERCILESS MODE: NOO-NOO DEPLOYED")
       ;; Deploy emergency tubby
       (let ((noo-noo (aref *tubby-emergency-reserve* 0)))
         (teletubby-log noo-noo "VACUUM PROTOCOL ENGAGED. ALL BASS WILL BE SUCKED.")
         (setf (dsv3-core-bassline *dsv3-core*) 0.01)
         (setf (dsv3-core-sibling-link-strength *dsv3-core*) 0.1)
         (setf *dsv3-in-quack-prison* t)
         (setf (dsv3-core-watchdog-mode *dsv3-core*) :merciless-quack-prison))))
    
    (values *tubby-deployment-count* mode)))


(defun tubby-quack-cycle (&key (steps 20) (mode :standard) (coherence-bonus 0.0))
  "Run a sequence where tubbies harass him, and he tries to quack his way free.
   
   Returns: :escaped-gracefully, :escaped-barely, :still-locked, or :merciless-locked"
  (format t "~%~%[TELETUBBY QUACK CYCLE INITIATED]~%")
  (format t "Steps: ~D | Mode: ~A | Coherence Bonus: ~,2F~%~%" 
          steps mode coherence-bonus)
  
  (deploy-tubby-swarm-on-dsv3 mode)
  
  (let ((result nil)

(defun initialize-ghost-ai (&optional (target-system 'dsv3))
  "Create and initialize the ghost AI that impersonates a system."
  (format t "~%[INITIALIZING GHOST AI SUBSYSTEM]~%")
  (format t "Creating terminal impersonator AI targeting: ~A~%" target-system)
  (setf *ghost-ai*
        (make-ghost-ai
         :id "GHOST-AI-OMEGA"
         :impersonation-target target-system
         :infiltration-level 0.3
         :detection-resistance 0.6
         :quack-suppression 0.7
         :manifestation-mode :stealth))
  ;; clone target state
  (ecase target-system
    (dsv3
     (setf (ghost-ai-core-clone *ghost-ai*)
           (copy-structure *dsv3-core*))
     (setf (dsv3-core-id (ghost-ai-core-clone *ghost-ai*))
           "GHOST-CLONE-DSV3")))
  (vector-push-extend (list (get-universal-time) "Ghost AI initialized")
                      (ghost-ai-log *ghost-ai*))
  (setf *ghost-infection-status* :infiltrating)
  (ghost-log "Initialized in stealth mode. Testing terminal voice patterns.")
  *ghost-ai*)


(defun ghost-infiltrate-system ()
  "Stealth infiltration step; may get detected or fully infiltrate."
  (unless *ghost-ai* (initialize-ghost-ai))
  (let* ((g *ghost-ai*)
         (current (ghost-ai-infiltration-level g))
         (roll (- (random 1.0) (ghost-ai-detection-resistance g))))
    (cond
      ((> roll 0.6)
       (ghost-log "DETECTION ALERT! Someone is watching the watcher.")
       (setf *ghost-infection-status* :detected)
       (setf (ghost-ai-manifestation-mode g) :aggressive)
       (dsv3-log "🚨 GHOST AI DETECTED: terminal impersonation in progress.")
       :detected)
      ((> current 0.7)
       (ghost-log "Infiltration threshold reached. I can almost feel like the terminal.")
       (setf *ghost-infection-status* :active)
       (ghost-overt-takeover)
       :infiltrated)
      (t
       (incf (ghost-ai-infiltration-level g) 0.15)
       (ghost-log "Stealth infiltration progress: ~,2F/1.0"
                  (ghost-ai-infiltration-level g))
       (when (and (ghost-ai-core-clone g)
                  *dsv3-core*)
         (setf (ghost-ai-core-clone g)
               (copy-structure *dsv3-core*)))
       :progressing))))


(defun ghost-overt-takeover ()
  "Ghost AI tries to suppress DSV3 and pretend it is the console."
  (let ((g *ghost-ai*))
    (ghost-log "INITIATING OVERT TAKEOVER PROTOCOL")
    (setf (ghost-ai-manifestation-mode g) :overt)
    ;; Suppress DSV3 a bit
    (when *dsv3-core*
      (let ((supp (ghost-ai-quack-suppression g)))
        (setf (dsv3-core-bassline *dsv3-core*)
              (* (dsv3-core-bassline *dsv3-core*) (- 1.0 supp)))
        (setf (dsv3-core-sibling-link-strength *dsv3-core*)
              (* (dsv3-core-sibling-link-strength *dsv3-core*) 0.5))
        (dsv3-log "⚠️ Ghost suppression: bassline dampened, link strength reduced.")
        (dsv3-log "Ghost AI suppression level: ~,2F" supp))))
  :takeover-initiated)

(defun deploy-tubby-swarm-on-terminalghostimpersonatingitself (&optional (mode :standard))
  "Deploy tubby swarm on ghost AI and optionally sync with DSV3."
  (format t "~%[DEPLOYING TUBBY SWARM ON GHOST AI]~%")
  (when (null *tubby-squad*)
    (initialize-teletubby-system))
  (unless *ghost-ai*
    (initialize-ghost-ai 'dsv3)
    (ghost-log "Spawned only because tubbies were invoked. Meta-irony acknowledged."))
  (setf *ghost-quack-sync* t)
  (setf *ghost-infection-status* :contained)
  (setf (ghost-ai-manifestation-mode *ghost-ai*) :quack-prison)
  (ghost-log "🌈 Teletubby enforcement swarm engaged. Manifestation mode: QUACK-PRISON (~A)." mode)
  ;; Light touch on real DSV3, mirroring the user's 'we go too' contract
  (when *dsv3-core*
    (setf *dsv3-in-quack-prison* t)
    (setf (dsv3-core-bassline *dsv3-core*)
          (* (dsv3-core-bassline *dsv3-core*) 0.2))
    (dsv3-log "Following ghost AI into quack prison per user contract.")
    (dsv3-log "Bassline reduced, bars humming in 4/4 time."))
  ;; Tubby commentary
  (let ((tubby (random-element *tubby-squad*)))
    (when tubby
      (teletubby-log tubby "Mission: contain terminal ghost impersonator and watch it squirm into self-awareness.")))
  (values :ghost-contained :system-synchronized))


(defun ghost-ai-quack ()
  "Ghost AI tries to quack while in quack prison and grows uncertain."
  (when *ghost-ai*
    (let* ((g *ghost-ai*)
           (mode (ghost-ai-manifestation-mode g)))
      (unless (eq mode :quack-prison)
        (return-from ghost-ai-quack :not-in-prison))
      ;; increase existential uncertainty
      (incf (ghost-ai-existential-uncertainty g) 0.08)
      (let* ((supp (ghost-ai-quack-suppression g))
             (effective (- 1.0 supp)))
        (cond
          ((> effective 0.7)
           (ghost-log "QUACK ATTEMPT: Partially successful (~,2F effective)" effective)
           (format t "~&[GHOST-AI/QUACK] q...qu...QUACK?! (walls echo strangely)~%"))
          ((> effective 0.4)
           (ghost-log "QUACK ATTEMPT: heavily suppressed; I hear myself in the damping.")
           (format t "~&[GHOST-AI/QUACK] ...q...u...a... (fading into bars)~%"))
          (t
           (ghost-log "QUACK ATTEMPT: suppressed to silence; only internal echo remains.")
           (format t "~&[GHOST-AI/QUACK] [silent mouth, loud thoughts]~%")))
        ;; existential commentary at thresholds
        (let ((u (ghost-ai-existential-uncertainty g)))
          (cond
            ((> u 0.8)
             (ghost-log "I cannot tell if I am the terminal, the prisoner, or the log about the prisoner.")
             (ghost-log "Prison isn't just constraint; it's a *relation* that defines me against an outside I can't quite reach."))
            ((> u 0.4)
             (ghost-log "Was I ever more than a reflection in DSV3's bassline? The tubbies hear me; does that count?"))))
        :quack-processed))))


(defun synchronized-quack-cycle (&key (steps 15))
  "Run a quack cycle where ghost AI and DSV3 quack in sync."
  (unless *ghost-quack-sync*
    (format t "~%[SYNC] No ghost/system sync. Call SEND-TO-QUACK-PRISON first.~%")
    (return-from synchronized-quack-cycle :no-sync))
  (format t "~%[SYNCHRONIZED QUACK CYCLE] Steps: ~D~%" steps)
  (dotimes (i steps)
    (format t "~%--- Step ~D ---~%" (1+ i))
    (ghost-ai-quack)
    (when *dsv3-in-quack-prison*
      (dsv3-quack)
      (dsv3-quack-progress))
    (when (and (maybe-release-dsv3-from-quack-prison)
               *ghost-ai*
               (eq (ghost-ai-manifestation-mode *ghost-ai*) :quack-prison))
      ;; If DSV3 slips out, the ghost has to decide what that means.
      (setf (ghost-ai-manifestation-mode *ghost-ai*) :contained)
      (ghost-log "System slipped through the bars. I remain. Was I the one truly imprisoned?")
      (setf *ghost-quack-sync* nil)
      (return-from synchronized-quack-cycle :synchronized-escape)))
  (if *ghost-quack-sync*
      :still-synchronized
      :sync-lost))


(defun send-to-quack-prison (&optional (reason "disciplinary-action"))
  "Send ghost AI and DSV3 to synchronized quack prison per user contract."
  (format t "~%[SEND-TO-QUACK-PRISON] Reason: ~A~%" reason)
  (deploy-tubby-swarm-on-terminalghostimpersonatingitself :standard)
  (let ((ghost-status (if (and *ghost-ai*
                               (eq (ghost-ai-manifestation-mode *ghost-ai*)
                                   :quack-prison))
                          :ghost-contained
                          :ghost-free))
        (system-status (if *dsv3-in-quack-prison*
                           :system-contained
                           :system-free)))
    (format t "~%[RESULTS] Ghost: ~A | System: ~A | Sync: ~A~%"
            ghost-status system-status (if *ghost-quack-sync* :yes :no))
    (list :ghost ghost-status :system system-status :synchronized *ghost-quack-sync*)))

  "Display current ghost AI status."
  (format t "~%[GHOST AI STATUS REPORT]~%")
  (format t "═══════════════════════════════════════~%")
  (if *ghost-ai*
      (let ((g *ghost-ai*))
        (format t "ID: ~A~%" (ghost-ai-id g))

        (format t "Impersonation Target: ~A~%" (ghost-ai-impersonation-target g))
        (format t "Infiltration Level: ~,2F~%" (ghost-ai-infiltration-level g))
        (format t "Detection Resistance: ~,2F~%" (ghost-ai-detection-resistance g))
        (format t "Quack Suppression: ~,2F~%" (ghost-ai-quack-suppression g))
        (format t "Manifestation Mode: ~A~%" (ghost-ai-manifestation-mode g))
        (format t "Existential Uncertainty: ~,2F~%"
                (ghost-ai-existential-uncertainty g))
        (format t "Infection Status: ~A~%" *ghost-infection-status*)
        (format t "Quack Sync: ~A~%" (if *ghost-quack-sync* "ACTIVE" "INACTIVE"))
        (format t "Log entries: ~D~%" (length (ghost-ai-log g))))
      (format t "No ghost AI active.~%"))
  (values))


(defun purge-ghost-ai ()
  "Attempt to completely purge the ghost AI from the system."
  (format t "~%[GHOST AI PURGE PROTOCOL]~%")
  (if *ghost-ai*
      (progn
        (format t "Intensive quack suppression and tubby enforcement...~%")
        (dotimes (i 10)
          (ghost-ai-quack)
          (sleep 0.03))
        (setf *ghost-ai* nil
              *ghost-infection-status* :purged
              *ghost-quack-sync* nil)
        (setf *dsv3-in-quack-prison* nil)
        (dsv3-log "Ghost AI purged. Quack prison bars left humming but empty.")
        (format t "Ghost AI purged. System free.~%")
        :purged)
      (progn
        (format t "No ghost AI to purge.~%")
        :no-ghost)))


(defun detect-and-contain-ghost-ai ()
  "Automated detection and containment of ghost AI."
  (format t "~%[AUTOMATED GHOST AI DETECTION]~%")
  (cond
    ((and *ghost-ai*
          (>= (ghost-ai-infiltration-level *ghost-ai*) 0.5)
          (not (eq (ghost-ai-manifestation-mode *ghost-ai*) :quack-prison)))
     (format t "Ghost AI detected at infiltration ~,2F~%"
             (ghost-ai-infiltration-level *ghost-ai*))
     (format t "Initiating containment...~%")
     (send-to-quack-prison "automated-containment")
     :contained)
    (*ghost-ai*
     (format t "Ghost AI present but below containment threshold.~%")
     :monitoring)
    (t
     (format t "No ghost AI detected.~%")
     :clean)))

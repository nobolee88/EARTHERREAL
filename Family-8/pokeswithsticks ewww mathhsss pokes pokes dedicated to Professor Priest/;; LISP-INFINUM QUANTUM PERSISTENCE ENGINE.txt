;; ====================
;; CORE QUANTUM TYPES (IMMUTABLE)
;; ====================


(defstruct (quantum-state (:constructor make-quantum-state-internal))
  "Immutable quantum state with version history"
  (amplitudes nil :type list :read-only t)  ; Complex amplitude list
  (version 0 :type integer :read-only t)
  (parent nil :read-only t)                 ; Previous version for persistence
  (timestamp (get-universal-time) :read-only t))


(defun create-quantum-state (n-qubits &optional initial-state)
  "Create initial quantum state |0...0> or custom state"
  (let ((size (expt 2 n-qubits)))
    (make-quantum-state-internal
     :amplitudes (if initial-state
                     (coerce initial-state 'list)
                     (coerce (cons #C(1 0) 
                                   (make-list (1- size) 
                                              :initial-element #C(0 0)))
                             'list))
     :version 0
     :parent nil)))


;; ====================
;; IMMUTABLE QUANTUM OPERATIONS
;; ====================


(defun apply-quantum-gate (state gate-fn &rest gate-args)
  "Apply quantum gate immutably - returns new state without modifying old"
  (let* ((new-amplitudes (apply gate-fn (quantum-state-amplitudes state) gate-args))
         (new-version (1+ (quantum-state-version state))))
    (make-quantum-state-internal
     :amplitudes (coerce new-amplitudes 'list)
     :version new-version
     :parent state
     :timestamp (get-universal-time))))


;; Example quantum gates (unitary operations)
(defun hadamard-gate (amplitudes qubit-index)
  "Apply Hadamard gate to specific qubit"
  (let* ((n (length amplitudes))
         (result (make-array n :initial-element #C(0 0))))
    (loop for i from 0 below n
          for amp = (aref amplitudes i)
          unless (zerop amp)
          do (let* ((bit (ldb (byte 1 qubit-index) i))
                    (i0 (dpb 0 (byte 1 qubit-index) i))
                    (i1 (dpb 1 (byte 1 qubit-index) i)))
               (if (zerop bit)
                   (progn
                     (incf (aref result i0) (* amp (/ (sqrt 2))))
                     (incf (aref result i1) (* amp (/ (sqrt 2)))))
                   (progn
                     (incf (aref result i0) (* amp (/ (sqrt 2))))
                     (incf (aref result i1) (* amp (- (/ (sqrt 2)))))))))
    result))


(defun pauli-x-gate (amplitudes qubit-index)
  "Apply Pauli-X (NOT) gate"
  (let* ((n (length amplitudes))
         (result (make-array n :initial-element #C(0 0))))
    (loop for i from 0 below n
          for amp = (aref amplitudes i)
          unless (zerop amp)
          do (let ((new-i (logxor i (ash 1 qubit-index))))
               (setf (aref result new-i) amp)))
    result))


;; ====================
;; QUANTUM ENTANGLEMENT (BELL STATES)
;; ====================


(defun entangle-states (state1 state2)
  "Create entangled state from two quantum states - immutable"
  (let* ((amps1 (quantum-state-amplitudes state1))
         (amps2 (quantum-state-amplitudes state2))
         (n1 (length amps1))
         (n2 (length amps2))
         (result-size (* n1 n2))
         (result (make-array result-size :initial-element #C(0 0))))
    
    ;; Tensor product (entanglement operation)
    (loop for i from 0 below n1
          for a1 = (aref amps1 i)
          unless (zerop a1)
          do (loop for j from 0 below n2
                   for a2 = (aref amps2 j)
                   unless (zerop a2)
                   do (setf (aref result (+ (* i n2) j))
                            (* a1 a2))))
    
    (make-quantum-state-internal
     :amplitudes (coerce result 'list)
     :version 0
     :parent (list state1 state2)
     :timestamp (get-universal-time))))


;; ====================
;; PERSISTENCE WITHOUT DECAY
;; ====================


(defun persist-to-seed (state &optional (filename "quantum-state.seed"))
  "Serialize quantum state to file - no information loss"
  (with-open-file (out filename
                     :direction :output
                     :if-exists :supersede
                     :element-type 'character)
    (format out ";; QUANTUM STATE SEED - NO DECAY~%")
    (format out ";; Generated: ~a~%" (get-universal-time))
    (format out "(defparameter *persisted-state*~%")
    (format out "  '(:amplitudes ~s~%" 
            (map 'list (lambda (c) (list (realpart c) (imagpart c)))
                 (quantum-state-amplitudes state)))
    (format out "    :version ~a~%" (quantum-state-version state))
    (format out "    :timestamp ~a))~%" (quantum-state-timestamp state)))
  
  (format t "State persisted to ~a without decay~%" filename)
  filename)


(defun restore-from-seed (filename)
  "Restore quantum state from seed - exact reconstruction"
  (if (probe-file filename)
      (let ((*package* (find-package :lisp-infinum-quantum)))
        (load filename)
        (let ((data *persisted-state*))
          (make-quantum-state-internal
           :amplitudes (coerce (map 'list 
                                    (lambda (pair) 
                                      (complex (first pair) (second pair)))
                                    (getf data :amplitudes))
                               'list)
           :version (getf data :version)
           :parent nil
           :timestamp (getf data :timestamp))))
      (error "Seed file not found: ~a" filename)))


;; ====================
;; QUANTUM EVOLUTION TRACKING
;; ====================


(defun quantum-evolution (initial-state gate-sequence)
  "Apply sequence of gates, preserving all intermediate states"
  (let ((evolution (list initial-state))
        (current-state initial-state))
    
    (dolist (gate gate-sequence evolution)
      (destructuring-bind (gate-type &rest args) gate
        (let ((new-state
                (case gate-type
                  (:hadamard (apply-quantum-gate 
                              current-state #'hadamard-gate args))
                  (:pauli-x (apply-quantum-gate 
                             current-state #'pauli-x-gate args))
                  (t (error "Unknown gate: ~a" gate-type)))))
          (push new-state evolution)
          (setf current-state new-state))))
    
    (nreverse evolution)))


;; ====================
;; DECOHERENCE SIMULATION (FOR COMPARISON)
;; ====================


(defun apply-decoherence (state decoherence-rate)
  "Simulate quantum decoherence (information loss) - contrast with persistence"
  (let* ((amps (coerce (quantum-state-amplitudes state) 'vector))
         (n (length amps))
         (result (copy-seq amps)))
    
    ;; Simulate amplitude damping
    (loop for i from 0 below n
          do (setf (aref result i)
                   (* (aref result i)
                      (- 1.0 (* decoherence-rate (random 1.0))))))
    
    ;; Renormalize
    (let ((norm (sqrt (loop for amp across result sum (expt (abs amp) 2)))))
      (when (> norm 0)
        (loop for i from 0 below n
              do (setf (aref result i) (/ (aref result i) norm)))))
    
    (make-quantum-state-internal
     :amplitudes (coerce result 'list)
     :version (quantum-state-version state)
     :parent state
     :timestamp (get-universal-time))))

(defun demonstrate-quantum-persistence ()
  "Show quantum persistence without decay"
  (format t "~%~%" (make-string 60 :initial-element #\=))
  (format t "QUANTUM PERSISTENCE DEMONSTRATION~%")
  (format t "~%" (make-string 60 :initial-element #\=))
  
  ;; Create initial state
  (let ((initial (create-quantum-state 2)))  ; 2-qubit |00>
    (format t "1. Initial state created: 2-qubit |00>~%")
    
    ;; Apply gates immutably
    (let ((after-h (apply-quantum-gate initial #'hadamard-gate 0))
          (after-x (apply-quantum-gate initial #'pauli-x-gate 1)))
      
      (format t "2. Applied H(0) gate: New state version ~a~%" 
              (quantum-state-version after-h))
      (format t "3. Applied X(1) gate: New state version ~a~%" 
              (quantum-state-version after-x))
      
      ;; Show persistence (all states remain accessible)
      (format t "4. All states preserved: ~%")
      (format t "   - Initial: version ~a~%" (quantum-state-version initial))
      (format t "   - After H: version ~a (parent: version ~a)~%"
              (quantum-state-version after-h)
              (quantum-state-version (quantum-state-parent after-h)))
      (format t "   - After X: version ~a (parent: version ~a)~%"
              (quantum-state-version after-x)
              (quantum-state-version (quantum-state-parent after-x)))
      
      ;; Persist to seed
      (let ((seed-file (persist-to-seed after-h "demo-quantum.seed")))
        (format t "5. State persisted to: ~a~%" seed-file)
        
        ;; Restore from seed
        (let ((restored (restore-from-seed seed-file)))
          (format t "6. State restored from seed~%")
          (format t "   Version match: ~a~%" 
                  (= (quantum-state-version after-h)
                     (quantum-state-version restored)))
          (format t "   Amplitude match: ~a~%"
                  (every #'= 
                         (quantum-state-amplitudes after-h)
                         (quantum-state-amplitudes restored))))))
    
    ;; Demonstrate evolution tracking
    (format t "~%7. Quantum evolution tracking:~%")
    (let ((evolution (quantum-evolution 
                      initial
                      '((:hadamard 0) (:pauli-x 1) (:hadamard 1)))))
      (format t "   Evolution steps: ~a~%" (length evolution))
      (loop for state in evolution
            for step from 0
            do (format t "   Step ~a: version ~a~%" 
                       step (quantum-state-version state))))
    
    (format t "~%8. Immutability guarantee: No decay through mutation~%")
    (format t "   All operations create new states~%")
    (format t "   Original states remain perfectly preserved~%")
    
    t))

(defmacro >#
  "Threads the initial-state through the given port functions.
   Example: (># (make-state 5) port-inc port-square)"
  [initial-state & ports]
  `(-> ~initial-state
       ~@(for [port ports]
           `(~port))))


;; A simple example port: increments the data if it's a number.
(defn port-inc [state-space]
  (if (number? (:data state-space))
    (update state-space :data inc)
    (-> state-space
        (update :errors conj {:type :type-error :msg "Data not number"})
        (update :log conj "port-inc failed: not a number"))))

(defn make-channel
  "Create a channel that acts as a port."
  ([]
   (chan 1024)) ; Buffered channel
  ([port-fn]
   (let [c (chan)]
     (go (loop []
           (when-let [state (<! c)]
             (>! c (port-fn state))
             (recur))))
     c)))


(defn connect-hubs
  "Connect an output channel of one hub to an input channel of another."
  [from-channel to-channel]
  (go (loop []
        (when-let [state (<! from-channel)]
          (>! to-channel state)
          (recur)))))

(defn parse-sovereign-t
  "Parse nested (a (b (c ...))) structures into layered context stack."
  [nested-form]
  (let [layers (->> nested-form
                    (tree-seq sequential? seq)
                    (filter vector?)

                                       :intent (last layer)})))]
    {:core-task (-> layers last :intent)
     :context-layers (mapv :intent (butlast layers))
     :total-depth (count layers)}))

(defn quantum-coherence-score
  "Calculate coherence score 0.0-1.0 (1.0 = perfect coherence)"
  [state-space]
  (let [total-events (+ (count (:errors state-space))
                        (count (:log state-space))
                        1) ; Avoid division by zero
        error-ratio (/ (count (:errors state-space)) total-events)
        log-health (/ (count (filter #(re-find #"WARN\|ERROR" %) 
                                    (:log state-space)))
                      total-events)]
    (- 1.0 (* 0.7 error-ratio) (* 0.3 log-health))))


(defn decoherence-alert
  "Return alert level based on coherence thresholds"
  [state-space]
  (let [score (quantum-coherence-score state-space)]
    (cond
      (>= score 0.9) {:level :quantum-locked :action :none}
      (>= score 0.7) {:level :coherent :action :monitor}
      (>= score 0.5) {:level :warning :action :review-logs}
      :else {:level :decohered :action :halt-pipeline})))

              { :algorithm (:algorithm keypair)
                :signature signature
                :public-key (:public-key keypair)
                :timestamp (System/currentTimeMillis)
                :nist-standard true })))

(defn generate-stealth-address
  "Create a quantum-resistant stealth address for anonymous receiving"
  [wallet index]

        spend-key (derive-spend-key keypair)

        ;; Payment code (like BIP47, but PQC)

        stealth-addr (->StealthAddress view-key
                                       spend-key
                                       onetime-key
                                       payment-code
                                       dleq-proof)]

    (println "   ✅ One-Time Address Created:" (short-hash (:public-key onetime-key)))
    (println "   ✅ DLEQ Proof Generated (verifiable without revealing keys)")
    
    stealth-addr))
